## 六、两个具名区块函数的签名与语义（**下一手派单照这条**）

所在件：`packages/skill-calorie/src/diet/nutritionPortDocs.ts`（深路径 import，不经 `diet/index.ts`）。

```ts
export function buildNutritionRatioBlock(v: NutritionRatioView, opts?: { copy?: boolean; command?: string }): string; // 交 #273，用在 calorie.view.diet-review 那张页
export function buildDietOverviewBlock(v: DietOverviewView, command?: string): string;                              // 交 #271，用在 calorie.view.diet 那张页
```
