// t489 独立复核探针（review seat 自设，被审 anchor-probe.mjs 覆盖不到的三处盲区）：
// 盲区1：被审脚本只数精确串 M1/M2-ANCHOR（git grep，只扫跟踪件）；本探针扫残留标记类
// M\d+-ANCHOR（比票面更宽）＋ src/**/*.ts 行尾 //…ANCHOR 注释形，范围含未跟踪件与 dist。
// 盲区2：被审脚本不看编译产物 dist；本探针查 dist 有无残留（无 dist 则记 NODIST）。
// 盲区3：被审脚本不证“只删了注释”；本探针对 317/392 去尾注释后代码与写死真值逐字比对。
import { readFileSync, existsSync, readdirSync, statSync } from 'node:fs';
import { join } from 'node:path';

const PKG = 'packages/skill-calorie';
const BODY = join(PKG, 'src/fetch/body.ts');
// 写死真值（取自 ea0a7e0^ 父版去尾注释后的代码，已与 git show 逐字核对）：
const EXPECT_317 = `  const cols = ['id', 'date', 'source', 'body_fat_pct', ...CALIPER_FIELDS, 'note'];`;
const EXPECT_392 = `    GROUP BY source, date ORDER BY date ASC\`).all(...clause.params) as unknown as`;

function classify(f, ln) {
  if (/M\d+-ANCHOR/.test(ln)) return 'red';
  if (f.includes(join(PKG, 'src')) && /\/\/.*ANCHOR/.test(ln)) return 'red';
  if (/-ANCHOR|\bANCHOR\b/.test(ln)) return 'advisory';
  return null;
}
function walk(dir, out) {
  for (const e of readdirSync(dir)) {
    const p = join(dir, e);
    const s = statSync(p);
    if (s.isDirectory()) { if (e !== 'node_modules') walk(p, out); }
    else if (/\.(ts|mjs|cjs|js)$/.test(e)) out.push(p);
  }
}
let redHits = [], advisoryHits = [];
function tag(f, i, ln) {
  const c = classify(f, ln);
  if (c === 'red') redHits.push(`${f}:${i + 1}:${ln.trim().slice(0, 80)}`);
  else if (c === 'advisory') advisoryHits.push(`${f}:${i + 1}:${ln.trim().slice(0, 80)}`);
}
for (const d of [join(PKG, 'src'), join(PKG, 'test'), join(PKG, 'scripts')]) {
  if (!existsSync(d)) continue;
  const files = [];
  walk(d, files);
  for (const f of files) {
    const txt = readFileSync(f, 'utf8');
    txt.split('\n').forEach((ln, i) => tag(f, i, ln));
  }
}
let distState = 'NODIST';
if (existsSync(join(PKG, 'dist'))) {
  distState = 'DIST-CHECKED';
  const files = [];
  try { walk(join(PKG, 'dist'), files); } catch { files.length = 0; }
  for (const f of files) {
    let txt = '';
    try { txt = readFileSync(f, 'utf8'); } catch { continue; }
    txt.split('\n').forEach((ln, i) => tag(f, i, ln));
  }
}
const lines = readFileSync(BODY, 'utf8').split('\n');
const strip = (s) => s.replace(/\s*\/\/.*$/, '').replace(/\s+$/, '');
const got317 = lines[316] ?? '';
const got392 = lines[391] ?? '';
const codeEq = strip(got317) === EXPECT_317 && strip(got392) === EXPECT_392;
const noTrailingAnchorComment = !/\/\/.*(-ANCHOR|\bANCHOR\b)/.test(got317) && !/\/\/.*(-ANCHOR|\bANCHOR\b)/.test(got392);

console.log(`REVIEW-PROBE RED_HITS=${redHits.length} ADVISORY=${advisoryHits.length} ${distState}`);
for (const h of redHits.slice(0, 10)) console.log(`  RED ${h}`);
for (const h of advisoryHits.slice(0, 10)) console.log(`  ADVISORY ${h}`);
console.log(`REVIEW-PROBE CODEEQUIV=${codeEq ? 'yes' : 'NO'} NOTRAIL=${noTrailingAnchorComment ? 'yes' : 'NO'}`);
console.log(`REVIEW-PROBE LINE317=[${strip(got317).slice(0, 60)}...] LINE392=[${strip(got392).slice(0, 60)}...]`);
const ok = redHits.length === 0 && codeEq && noTrailingAnchorComment;
console.log(ok ? 'REVIEW-GREEN: yes (residue-class zero + code identical)' : 'REVIEW-RED: yes (residue or code drift)');
process.exit(ok ? 0 : 1);
