- 实现：`D:\2Study\StudyNotes\SKILLS\卡路里\templates\body_composition_wizard.html:434-450`（函数 `jp7` 起于 `:438`）：

```
// ---------- Jackson-Pollock 7 点法 ----------
// 男:BD = 1.112 - 0.00043499*sum + 0.00000055*sum^2 - 0.00028826*age
// 女:BD = 1.097 - 0.00046971*sum + 0.00000056*sum^2 - 0.00012828*age
// %fat = (495 / BD) - 450
function jp7(sumMm, age, sex) {
  if (!age || !sex) return null;
  const sum = +sumMm, a = +age;
  let bd;
  if (sex === 'male') {
    bd = 1.112 - 0.00043499 * sum + 0.00000055 * sum * sum - 0.00028826 * a;
  } else if (sex === 'female') {
    bd = 1.097 - 0.00046971 * sum + 0.00000056 * sum * sum - 0.00012828 * a;
  } else return null;
  if (!(bd > 0)) return null;
  const pct = (495 / bd) - 450;
  return Math.round(pct * 100) / 100;
}
```

- 文字口径：`D:\2Study\StudyNotes\SKILLS\卡路里\SKILL.md:1173-1176`
